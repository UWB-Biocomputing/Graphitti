212 T getPastEvent(int offset) const
213 {
214 // Quick checks: offset must be in past, and not larger than the buffer size
215 assert(((offset < 0)) && (offset > -(this->dataSeries_.size() - 1)));
216
217 // The event is at bufferEnd_ + offset (taking into account the
218 // buffer size, and the fact that offset is negative).
219 int index = bufferEnd_ + offset;
220 if (index < 0)
221 index += this->dataSeries_.size();
222
223 // Need to check that we're not asking for an item so long ago that it is
224 // not in the buffer. Note that there are three possibilities:
225 // 1. if bufferEnd_ > bufferFront_, then valid entries are within the range
226 // [bufferFront_, bufferEnd_)
227 // 2. if bufferEnd_ < bufferFront_, then the buffer wraps around the end of
228 // vector and valid entries are within the range [0, bufferEnd_) or the
229 // range [bufferFront_, size()).
230 // 3. if buffer is empty (bufferFront_ == bufferEnd_), then there are no events
231 //
232 // Note that this means that index at this point must always be less than
233 // bufferEnd_ AND >= queueFront.
234 if ((index < bufferEnd_) && (index >= bufferFront_))
235 return this->dataSeries_[index];
236 else
237 return numeric_limits<T>::max();
238 }
