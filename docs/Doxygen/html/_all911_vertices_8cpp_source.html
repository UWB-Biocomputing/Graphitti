55// Creates all the Vertices and assigns initial data for them.
56void All911Vertices::createAllVertices(Layout &layout)
57{
58 // Read Input Events using the InputManager
59 inputManager_.readInputs();
60 LOG4CPLUS_DEBUG(vertexLogger_,
61 "Total number of events: " << inputManager_.getTotalNumberOfEvents());
62
63 Simulator &simulator = Simulator::getInstance();
64 // For metrics whose entries are recorded for each time step such as queue length history
65 uint64_t stepsPerEpoch = simulator.getEpochDuration() / simulator.getDeltaT();
66 // For metrics whose entries are recorded for each call such as begin time history
67 int maxEventsPerEpoch = static_cast<int>(Simulator::getInstance().getEpochDuration()
68 * Simulator::getInstance().getMaxFiringRate());
69 LOG4CPLUS_DEBUG(vertexLogger_, "Steps per epoch: " << stepsPerEpoch);
70 LOG4CPLUS_DEBUG(vertexLogger_, "Max events per epoch: " << maxEventsPerEpoch);
71
72 // Loop over all vertices and set the number of servers and trunks.
73 // We get the information needed from the GraphManager.
74 GraphManager<NG911VertexProperties>::VertexIterator vi, vi_end;
75 GraphManager<NG911VertexProperties> &gm = GraphManager<NG911VertexProperties>::getInstance();
76 // Variable to map caller region IDs to 0 to n-1 where n is the number of caller regions.
77 // This allows caller regions to exist anywhere in the graph file
78 numberOfVerticesNeedingDeviceNoise_ = 0;
79 for (boost::tie(vi, vi_end) = gm.vertices(); vi != vi_end; ++vi) {
80 assert(*vi < size_);
81
82 if (gm[*vi].type == "CALR") {
83 vertexType_[*vi] = 3;
84 vertexIdToNoiseIndex_[*vi] = numberOfVerticesNeedingDeviceNoise_;
85 numberOfVerticesNeedingDeviceNoise_++;
86 } else {
87 if (gm[*vi].type == "PSAP") {
88 vertexType_[*vi] = 4;
89 } else if (gm[*vi].type == "EMS") {
90 vertexType_[*vi] = 5;
91 } else if (gm[*vi].type == "FIRE") {
92 vertexType_[*vi] = 6;
93 } else if (gm[*vi].type == "LAW") {
94 vertexType_[*vi] = 7;
95 }
96 numServers_[*vi] = gm[*vi].servers;
97 numTrunks_[*vi] = gm[*vi].trunks;
98 // We should not have more servers than trunks
99 assert(numServers_[*vi] <= numTrunks_[*vi]);
100 if (maxNumberOfServers_ < numServers_[*vi]) {
101 maxNumberOfServers_ = numServers_[*vi];
102 }
103 }
104 }
105
106 LOG4CPLUS_DEBUG(vertexLogger_, "Number of vertices needing device noise: "
107 << numberOfVerticesNeedingDeviceNoise_);
108 LOG4CPLUS_DEBUG(vertexLogger_, "Max number of servers: " << maxNumberOfServers_);
109
110 // Loop over the vertices again to appropriate resize data members such that
111 // each data member used the same size for all of it's vertices. This is to
112 // help with mirroring the implementation on the GPU where we need a consistent
113 // size for all vertices.
114 for (int vertexId = 0; vertexId < size_; vertexId++) {
115 // Initialize the data structures for system metrics
116 beginTimeHistory_[vertexId].resize(maxEventsPerEpoch);
117 answerTimeHistory_[vertexId].resize(maxEventsPerEpoch);
118 endTimeHistory_[vertexId].resize(maxEventsPerEpoch);
119 wasAbandonedHistory_[vertexId].resize(maxEventsPerEpoch);
120 queueLengthHistory_[vertexId].resize(stepsPerEpoch);
121 utilizationHistory_[vertexId].resize(stepsPerEpoch);
122 vertexQueues_[vertexId].resize(stepsPerEpoch);
123 // Initialize the data structures for agent availability
124 servingCall_[vertexId].resize(maxNumberOfServers_);
125 answerTime_[vertexId].resize(maxNumberOfServers_);
126 serverCountdown_[vertexId].assign(maxNumberOfServers_, 0);
127 }
128}

236// Take calls from the edges and transfer them to the vertex if it's queue is not full
237void All911Vertices::integrateVertexInputs(AllEdges &edges, EdgeIndexMap &edgeIndexMap)
238{
239 Simulator &simulator = Simulator::getInstance();
240 All911Edges &all911Edges = dynamic_cast<All911Edges &>(edges);
241
242 for (int vertex = 0; vertex < simulator.getTotalVertices(); ++vertex) {
243 int start = edgeIndexMap.incomingEdgeBegin_[vertex];
244 int count = edgeIndexMap.incomingEdgeCount_[vertex];
245
246 if (simulator.getModel().getLayout().vertexTypeMap_[vertex] == vertexType::CALR) {
247 continue; // TODO911: Caller Regions will have different behaviour
248 }
249
250 // Loop over all the edges and pull the data in
251 for (int edge = start; edge < start + count; ++edge) {
252 int edgeIdx = edgeIndexMap.incomingEdgeIndexMap_[edge];
253
254 if (!all911Edges.inUse_[edgeIdx] || all911Edges.isAvailable_[edgeIdx]) {
255 continue; // Edge isn't in use and doesn't have a call
256 }
257
258 int dst = all911Edges.destVertexIndex_[edgeIdx];
259 // The destination vertex should be the one pulling the information
260 assert(dst == vertex);
261
262 CircularBuffer<Call> &dstQueue = getQueue(dst);
263 // Compute the size of the destination queue
264 // Allows us to use larger capacity queues but treat them like they are smaller
265 // to simplify the mirroring on the GPU.
266 uint64_t dstQueueSize;
267 uint64_t queueFrontIndex = dstQueue.getFrontIndex();
268 uint64_t queueEndIndex = dstQueue.getEndIndex();
269 if (queueFrontIndex >= queueEndIndex) {
270 dstQueueSize = queueFrontIndex - queueEndIndex;
271 } else {
272 // Internal CircularBuffer buffer size is capacity + 1
273 dstQueueSize = numTrunks_[dst] + 1 + queueFrontIndex - queueEndIndex;
274 }
275
276 // Compute the capacity of the destination queue
277 int dstQueueCapacity = numTrunks_[dst];
278 // dstQueueSize can't be negative but we need to be able to compare it to a possible negative waiting queue
279 // so cast the size to an int for comparison
280 if ((int)dstQueueSize >= (dstQueueCapacity - busyServers(dst))) {
281 // Call is dropped because there is no space in the waiting queue
282 if (!all911Edges.isRedial_[edgeIdx]) {
283 // Only count the dropped call if it's not a redial
284 droppedCalls(dst)++;
285 // Record that we received a call
286 receivedCalls(dst)++;
287 LOG4CPLUS_DEBUG(vertexLogger_, "Call dropped: " << droppedCalls(dst) << ", time: "
288 << all911Edges.call_[edgeIdx].time
289 << ", vertex: " << dst
290 << ", queue size: " << dstQueueSize);
291 }
292 } else {
293 // Transfer call to destination
294 assert(((queueFrontIndex + 1) % numTrunks_[dst] + 1) != queueEndIndex);
295 vector<Call> &queueBuffer = dstQueue.getBuffer();
296 queueBuffer[queueFrontIndex] = all911Edges.call_[edgeIdx];
297 uint64_t newFrontIndex = (queueFrontIndex + 1) % (numTrunks_[dst] + 1);
298 dstQueue.setFrontIndex(newFrontIndex);
299 // Record that we received a call
300 receivedCalls(dst)++;
301 all911Edges.isAvailable_[edgeIdx] = true;
302 all911Edges.isRedial_[edgeIdx] = false;
303 }
304 }
305 }

308// Update internal state of the indexed vertex (called by every simulation step).
309void All911Vertices::advanceVertices(AllEdges &edges, const EdgeIndexMap &edgeIndexMap)
310{
311 Simulator &simulator = Simulator::getInstance();
312 Layout &layout = simulator.getModel().getLayout();
313
314 All911Edges &edges911 = dynamic_cast<All911Edges &>(edges);
315
316 // Advance vertices
317 for (int vertex = 0; vertex < simulator.getTotalVertices(); ++vertex) {
318 if (layout.vertexTypeMap_[vertex] == vertexType::CALR) {
319 advanceCALR(vertex, edges911, edgeIndexMap);
320 } else if (layout.vertexTypeMap_[vertex] == vertexType::PSAP) {
321 advancePSAP(vertex, edges911, edgeIndexMap);
322 } else if (layout.vertexTypeMap_[vertex] == vertexType::EMS
323 || layout.vertexTypeMap_[vertex] == vertexType::FIRE
324 || layout.vertexTypeMap_[vertex] == vertexType::LAW) {
325 advanceRESP(vertex, edges911, edgeIndexMap);
326 }
327 }

331// Advance a CALR vertex. Send calls to the appropriate PSAP
332void All911Vertices::advanceCALR(BGSIZE vertexIdx, All911Edges &edges911,
333 const EdgeIndexMap &edgeIndexMap)
334{
335 // // There is only one outgoing edge from CALR to a PSAP
336 BGSIZE edgeIdx = edgeIndexMap.outgoingEdgeIndexMap_[edgeIndexMap.outgoingEdgeBegin_[vertexIdx]];
337
338 unsigned char makeAvailable = (1 - edges911.isAvailable_[edgeIdx])
339 * (1 - edges911.isRedial_[edgeIdx])
340 * (unsigned char)(initRNG.randDblExc() >= redialP_);
341
342 edges911.isAvailable_[edgeIdx] |= makeAvailable;
343 edges911.isRedial_[edgeIdx] |= (1 - edges911.isAvailable_[edgeIdx]) * (1 - makeAvailable);
344
345 // We can use CircularBuffer methods because we don't need the caller region queue
346 // to behave like it only has a capacity of numTrunks_ like we do for other vertices.
347 //
348 // peek at the next call in the queue
349 optional<Call> nextCall = vertexQueues_[vertexIdx].peek();
350 if (edges911.isAvailable_[edgeIdx] && nextCall && nextCall->time <= g_simulationStep) {
351 // Calls that start at the same time are process in the order they appear.
352 // The call starts at the current time step so we need to pop it and process it
353 vertexQueues_[vertexIdx].get(); // pop from the queue
354
355 // Place new call in the edge going to the PSAP
356 assert(edges911.isAvailable_[edgeIdx]);
357 edges911.call_[edgeIdx] = nextCall.value();
358 edges911.isAvailable_[edgeIdx] = false;
359 LOG4CPLUS_DEBUG(vertexLogger_, "Calling PSAP at time: " << nextCall->time);
360 }
361}

364// Advance a PSAP vertex. Controls the redirection and handling of calls.
365void All911Vertices::advancePSAP(BGSIZE vertexIdx, All911Edges &edges911,
366 const EdgeIndexMap &edgeIndexMap)
367{
368 int numberOfServers = numServers_[vertexIdx];
369 // Loop over all servers and free the ones finishing serving calls
370 int numberOfAvailableServers = 0;
371 vector<unsigned char>
372 availableServers; // Use vector but treat like array to better mirror on GPU
373 availableServers.reserve(numberOfServers);
374 // Initialize to no servers having been assigned a call yet
375 for (BGSIZE serverIndex = 0; serverIndex < numberOfServers; serverIndex++) {
376 availableServers[serverIndex] = false;
377 }
378 for (size_t server = 0; server < numberOfServers; ++server) {
379 int countdown = serverCountdown_[vertexIdx][server];
380 // Check if countdown was already 0
381 int countdownWasZero = countdown == 0;
382
383 // Decrement if it was not already 0
384 countdown -= (1 - countdownWasZero);
385 serverCountdown_[vertexIdx][server] = countdown;
386
387 // Set the available server if it was already available or became available
388 availableServers[server] = (unsigned char)(countdown == 0);
389 numberOfAvailableServers += (countdown == 0);
390
391 // If it became zero, the unit responds to the new incident
392 if ((!countdownWasZero) & (countdown == 0)) {
393 // Server becomes free to take calls
394 // TODO: What about wrap-up time?
395 Call &endingCall = servingCall_[vertexIdx][server];
396
397 //Store call metrics
398 wasAbandonedHistory_[vertexIdx].insertEvent(false);
399 beginTimeHistory_[vertexIdx].insertEvent(endingCall.time);
400 answerTimeHistory_[vertexIdx].insertEvent(answerTime_[vertexIdx][server]);
401 endTimeHistory_[vertexIdx].insertEvent(g_simulationStep);
402 LOG4CPLUS_DEBUG(vertexLogger_,
403 "Finishing call, begin time: "
404 << endingCall.time << ", end time: " << g_simulationStep
405 << ", waited: " << answerTime_[vertexIdx][server] - endingCall.time);
406
407 // Dispatch the Responder closest to the emergency location.
408 BGSIZE respEdge = getEdgeToClosestResponder(endingCall, vertexIdx);
409 BGSIZE responder = edges911.destVertexIndex_[respEdge];
410 LOG4CPLUS_DEBUG(vertexLogger_, "Dispatching Responder: " << responder);
411
412 // Place the call in the edge going to the responder
413 // Call becomes a dispatch order at this time
414 endingCall.time = g_simulationStep;
415 edges911.call_[respEdge] = endingCall;
416 edges911.isAvailable_[respEdge] = false;
417 }
418 }
419
420 // Need the initial number of servers for utilization metric as well as a number of servers that can change
421 // during the while loop iterations
422 int currentlyAvailableServers = numberOfAvailableServers;
423 // Assign calls to servers until either no servers are available or
424 // there are no more calls in the waiting queue
425 while (currentlyAvailableServers > 0 && !vertexQueues_[vertexIdx].isEmpty()) {
426 // TODO: calls with duration of zero are being added but because countdown will be zero
427 // they don't show up in the logs
428 //
429 // Internal CircularBuffer buffer size is capacity + 1
430 vector<Call> queueBuffer = vertexQueues_[vertexIdx].getBuffer();
431 uint64_t queueEnd = vertexQueues_[vertexIdx].getEndIndex();
432 Call call = queueBuffer[queueEnd];
433 uint64_t newEndIndex = (queueEnd + 1) % (numTrunks_[vertexIdx] + 1);
434 vertexQueues_[vertexIdx].setEndIndex(newEndIndex);
435
436 if (call.patience < (g_simulationStep - call.time)) {
437 // If the patience time is less than the waiting time, the call is abandoned
438 wasAbandonedHistory_[vertexIdx].insertEvent(true);
439 beginTimeHistory_[vertexIdx].insertEvent(call.time);
440 // Answer time and end time get zero as sentinel for non-valid values
441 answerTimeHistory_[vertexIdx].insertEvent(0);
442 endTimeHistory_[vertexIdx].insertEvent(0);
443 LOG4CPLUS_DEBUG(vertexLogger_, "Call was abandoned, Patience: "
444 << call.patience
445 << " Ring Time: " << g_simulationStep - call.time);
446 } else {
447 // The available server starts serving the call
448 int availServer;
449 for (BGSIZE serverIndex = 0; serverIndex < numberOfServers; serverIndex++) {
450 if (availableServers[serverIndex] == true) {
451 // If server is available, have that server serve the call
452 availServer = serverIndex;
453 availableServers[serverIndex] = false;
454 currentlyAvailableServers--;
455 break;
456 }
457 }
458 servingCall_[vertexIdx][availServer] = call;
459 answerTime_[vertexIdx][availServer] = g_simulationStep;
460 serverCountdown_[vertexIdx][availServer] = call.duration;
461 LOG4CPLUS_DEBUG(vertexLogger_, "Serving Call starting at time: "
462 << call.time << ", sim-step: " << g_simulationStep);
463 }
464 }
465
466 // Update number of busy servers. This is used to check if there is space in the queue
467 busyServers_[vertexIdx] = numberOfServers - numberOfAvailableServers;
468
469 // Update queueLength and utilization histories
470 uint64_t queueSize;
471 uint64_t queueFront = vertexQueues_[vertexIdx].getFrontIndex();
472 uint64_t queueEnd = vertexQueues_[vertexIdx].getEndIndex();
473 if (queueFront >= queueEnd) {
474 queueSize = queueFront - queueEnd;
475 } else {
476 // Internal CircularBuffer buffer size is capacity + 1
477 queueSize = numTrunks_[vertexIdx] + 1 + queueFront - queueEnd;
478 }
479 queueLengthHistory_[vertexIdx].insertEvent(queueSize);
480 utilizationHistory_[vertexIdx].insertEvent(static_cast<float>(busyServers_[vertexIdx])
481 / numberOfServers);
482}

485// Advance a RESP vertex. Receives call from PSAP and responds to the emergency events
486void All911Vertices::advanceRESP(BGSIZE vertexIdx, All911Edges &edges911,
487 const EdgeIndexMap &edgeIndexMap)
488{
489 Layout &layout = Simulator::getInstance().getModel().getLayout();
490 Layout911 &layout911 = dynamic_cast<Layout911 &>(layout);
491
492 // Free the units finishing up with emergency responses
493 int numberOfAvailableUnits = 0;
494 vector<unsigned char>
495 availableUnits; // Use vector but treat like array to better mirror on GPU
496 availableUnits.reserve(numServers_[vertexIdx]);
497 for (BGSIZE unitIndex = 0; unitIndex < numServers_[vertexIdx]; unitIndex++) {
498 availableUnits[unitIndex] = false;
499 }
500 for (size_t unit = 0; unit < numServers_[vertexIdx]; ++unit) {
501 int countdown = serverCountdown_[vertexIdx][unit];
502 // Check if countdown was already 0
503 int countdownWasZero = countdown == 0;
504
505 // Decrement if it was not already 0
506 countdown -= (1 - countdownWasZero);
507 serverCountdown_[vertexIdx][unit] = countdown;
508
509 // Set the available unit if it was already available or became available
510 availableUnits[unit] = (unsigned char)(countdown == 0);
511 numberOfAvailableUnits += (countdown == 0);
512
513 // If it became zero, the unit responds to the new incident
514 if ((!countdownWasZero) & (countdown == 0)) {
515 // Unit becomes available to responde to new incidents
516 Call &endingIncident = servingCall_[vertexIdx][unit];
517
518 //Store incident response metrics
519 wasAbandonedHistory_[vertexIdx].insertEvent(false);
520 beginTimeHistory_[vertexIdx].insertEvent(endingIncident.time);
521 answerTimeHistory_[vertexIdx].insertEvent(answerTime_[vertexIdx][unit]);
522 endTimeHistory_[vertexIdx].insertEvent(g_simulationStep);
523 LOG4CPLUS_DEBUG(vertexLogger_,
524 "Finishing response, begin time: "
525 << endingIncident.time << ", end time: " << g_simulationStep
526 << ", waited: " << answerTime_[vertexIdx][unit] - endingIncident.time);
527 }
528 }
529
530
531 // Assign reponse dispatches until no units are available or there are no more
532 // incidents in the waiting queue
533 for (size_t unit = 0; unit < numberOfAvailableUnits && !vertexQueues_[vertexIdx].isEmpty();
534 ++unit) {
535 // Internal CircularBuffer buffer size is capacity + 1
536 vector<Call> queueBuffer = vertexQueues_[vertexIdx].getBuffer();
537 uint64_t queueEnd = vertexQueues_[vertexIdx].getEndIndex();
538 Call incident = queueBuffer[queueEnd];
539 uint64_t newEndIndex = (queueEnd + 1) % (numTrunks_[vertexIdx] + 1);
540 vertexQueues_[vertexIdx].setEndIndex(newEndIndex);
541
542 // The available unit starts serving the call
543 int availUnit = -1;
544 for (BGSIZE unitIndex = 0; unitIndex < numServers_[vertexIdx]; unitIndex++) {
545 // Add 0 if unit is not available or 1 + unitIndex if it's available and a unit has not already been found
546 availUnit += (availableUnits[unitIndex] == true && availUnit == -1) * (unitIndex + 1);
547 // Flip value only if the unit is available and a unit has not been found
548 availableUnits[unitIndex]
549 = (unsigned char)(availableUnits[unitIndex]
550 == true - (availableUnits[unitIndex] == true && availUnit == -1));
551 }
552 servingCall_[vertexIdx][availUnit] = incident;
553 answerTime_[vertexIdx][availUnit] = g_simulationStep;
554
555 // We need to calculate the distance in miles but the x and y coordinates
556 // represent, respectively, degrees of longitude and latitude.
557 // One degree of latitude is aproximately 69 miles regardles of the location. However,
558 // a degree of longitude varies, being 69.172 miles at the equator and gradually shrinking
559 // to zero at the poles.
560 // One degree of longitude can be converted to miles using the following formula:
561 // 1 degree of longitude = cos(latitude) * 69.172
562 double lngDegreeLength = cos(layout911.yloc_[vertexIdx] * (pi / 180)) * 69.172;
563 double latDegreeLength = 69.0;
564 double deltaLng = incident.x - layout911.xloc_[vertexIdx];
565 double deltaLat = incident.y - layout911.yloc_[vertexIdx];
566 double dist2incident
567 = sqrt(pow(deltaLng * lngDegreeLength, 2) + pow(deltaLat * latDegreeLength, 2));
568
569 // Calculate the driving time to the incident in seconds
570 double driveTime = (dist2incident / avgDrivingSpeed_) * 3600;
571 serverCountdown_[vertexIdx][availUnit] = driveTime + incident.onSiteTime;
572
573 serverCountdown_[vertexIdx][availUnit] = incident.duration;
574 LOG4CPLUS_DEBUG(vertexLogger_, "Response, driving time: " << driveTime << ", On-site time: "
575 << incident.onSiteTime);
576 }
577
578 // Update number of busy servers. This is used to check if there is space in the queue
579 busyServers_[vertexIdx] = numServers_[vertexIdx] - numberOfAvailableUnits;
580
581 // Update queueLength and utilization histories
582 uint64_t queueSize;
583 uint64_t queueFront = vertexQueues_[vertexIdx].getFrontIndex();
584 uint64_t queueEnd = vertexQueues_[vertexIdx].getEndIndex();
585 if (queueFront >= queueEnd) {
586 queueSize = queueFront - queueEnd;
587 } else {
588 // Internal CircularBuffer buffer size is capacity + 1
589 queueSize = numTrunks_[vertexIdx] + 1 + queueFront - queueEnd;
590 }
591 queueLengthHistory_[vertexIdx].insertEvent(queueSize);
592 utilizationHistory_[vertexIdx].insertEvent(static_cast<float>(busyServers_[vertexIdx])
593 / numServers_[vertexIdx]);
594}

599BGSIZE All911Vertices::getEdgeToClosestResponder(const Call &call, BGSIZE vertexIdx)
600{
601 Connections &connections = Simulator::getInstance().getModel().getConnections();
602 All911Edges &edges911 = dynamic_cast<All911Edges &>(connections.getEdges());
603 EdgeIndexMap &edgeIndexMap = connections.getEdgeIndexMap();
604
605 vertexType requiredType;
606 if (call.type == "Law")
607 requiredType = vertexType::LAW;
608 else if (call.type == "EMS")
609 requiredType = vertexType::EMS;
610 else if (call.type == "Fire")
611 requiredType = vertexType::FIRE;
612
613 // loop over the outgoing edges looking for the responder with the shortest
614 // Euclidean distance to the call's location.
615 BGSIZE startOutEdg = edgeIndexMap.outgoingEdgeBegin_[vertexIdx];
616 BGSIZE outEdgCount = edgeIndexMap.outgoingEdgeCount_[vertexIdx];
617 Layout911 &layout911
618 = dynamic_cast<Layout911 &>(Simulator::getInstance().getModel().getLayout());
619
620 BGSIZE resp, respEdge;
621 double minDistance = numeric_limits<double>::max();
622 for (BGSIZE eIdxMap = startOutEdg; eIdxMap < startOutEdg + outEdgCount; ++eIdxMap) {
623 BGSIZE outEdg = edgeIndexMap.outgoingEdgeIndexMap_[eIdxMap];
624 assert(edges911.inUse_[outEdg]); // Edge must be in use
625
626 BGSIZE dstVertex = edges911.destVertexIndex_[outEdg];
627 if (layout911.vertexTypeMap_[dstVertex] == requiredType) {
628 double distance = layout911.getDistance(dstVertex, call.x, call.y);
629
630 if (distance < minDistance) {
631 minDistance = distance;
632 resp = dstVertex;
633 respEdge = outEdg;
634 }
635 }
636 }
637
638 // We must have found the closest responder of the right type
639 assert(minDistance < numeric_limits<double>::max());
640 assert(layout911.vertexTypeMap_[resp] == requiredType);
641 return respEdge;
642}
